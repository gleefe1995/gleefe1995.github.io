```c++
#include <iostream>
#include <cstdarg> // for ellipsis
using namespace std;

double findAverage(int count, ...){
	double sum=0;
	va_list list;
	
	va_start(list, count);
	for (int arg=0;arg<count;++arg)
		sum+=va_arg(list,int);
	
	va_end(list);
	
	return sum/count;
	
}

int main(){

	cout << findAverage(4,1,2,3,4) << "\n"; // 2.5
	cout << findAverage(10,1,2,3) << "\n"; // 이상한 숫자 나옴

	return 0;
}
```
