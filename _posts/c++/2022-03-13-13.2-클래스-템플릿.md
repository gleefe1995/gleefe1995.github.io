// class 자체를 explicit 해줄 수도 있다.

int main()
{
//    MyArray my_array(10);
 	MyArray<double> my_array(10); // T로 정의할 땐 데이터타입을 정의해줘야한다.
    
    for (int i=0; i<my_array.getLength(); ++i)
        my_array[i] = i * 0.5;
   	
    my_array.print();
    
    return 0;
}
