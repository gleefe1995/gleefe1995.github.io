#include <iostream>
using namespace std;

class Cents
{
private:
    int m_cents;

public:
    Cents(int cents=0) { m_cents = cents; }
    int getCents() const { return m_cents; }
    int& getCents() { return m_cents; }
    
    Cents operator - () const
    {
        return Cents(-m_cents);
    }
    
    bool operator ! () const
    {
        // !Cents(...) => 돈이 없으면 True로 정하는 등 자기가 정해야함.
        return (m_cents == 0) ? true : false;
    }
    
};

int main()
{
    Cents cents1(6);
    Cents cents2(0);
    
    cout << -cents1 << "\n";
    cout << -Cents(10) << "\n";
