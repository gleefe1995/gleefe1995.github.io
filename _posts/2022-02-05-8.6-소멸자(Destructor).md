class IntArray
{
private:
    int *m_arr = nullptr;
    int m_length = 0;
    
public:
    IntArray(const int length_in)
    {
        m_length = length_in;
        m_arr = new int[m_length];
    }
    
    ~IntArray()
    {
        if (m_arr !=nullptr) delete[] m_arr; // 소멸하면서 delete되도록 하면 memory 누수가 생기지 않음
    }
    
    int getLength(){ return m_length;}
    
}

int main()
{
    while (true)
    {
        IntArray my_int_arr(10000); // memory leak
        
    }
    return 0;
}
